    startup_type0 = model.addVars(periods, vtype=GRB.INTEGER, name="startup_type0")
    startup_type1 = model.addVars(periods, vtype=GRB.INTEGER, name="startup_type1")
    
    # Objective: minimize total cost
    fixed_costs = gp.quicksum(
        fixed_cost_type0 * operational_type0[t] + fixed_cost_type1 * operational_type1[t]
        for t in periods
    )
    
    variable_costs = gp.quicksum(
        variable_cost_type0 * power_type0[t] + variable_cost_type1 * power_type1[t]
        for t in periods
    )
    
    startup_costs = gp.quicksum(
        startup_cost_type0 * startup_type0[t] + startup_cost_type1 * startup_type1[t]
        for t in periods
    )
    
    model.setObjective(fixed_costs + variable_costs + startup_costs, GRB.MINIMIZE)
    
    # Constraints
    for t in periods:
        # Demand satisfaction
        model.addConstr(
            power_type0[t] + power_type1[t] >= demands[t],
            name=f"demand_{t}"
        )
        
        # Minimum output constraints (when operational)
        model.addConstr(
            power_type0[t] >= min_output_type0 * operational_type0[t],
            name=f"min_output_type0_{t}"
        )
        model.addConstr(
            power_type1[t] >= min_output_type1 * operational_type1[t],
            name=f"min_output_type1_{t}"
        )
        
        # Maximum output constraints (when operational)
        model.addConstr(
            power_type0[t] <= max_output_type0 * operational_type0[t],
            name=f"max_output_type0_{t}"
        )
        model.addConstr(
            power_type1[t] <= max_output_type1 * operational_type1[t],
            name=f"max_output_type1_{t}"
        )
        
        # Total capacity requirement
        model.addConstr(
            max_output_type0 * operational_type0[t] + max_output_type1 * operational_type1[t] >= max_capacity_req[t],
            name=f"capacity_req_{t}"
        )
        
        # Maximum operational generators
        model.addConstr(
            operational_type0[t] <= max_operational_type0,
            name=f"max_op_type0_{t}"
        )
        model.addConstr(
            operational_type1[t] <= max_operational_type1,
            name=f"max_op_type1_{t}"
        )
        
        # Startup constraints for period 0
        if t == 0:
            model.addConstr(
                startup_type0[t] <= max_startup_type0_period0,
                name=f"max_startup_type0_{t}"
            )
            model.addConstr(
                startup_type1[t] <= max_startup_type1_period0,
                name=f"max_startup_type1_{t}"
            )
            # For period 0, operational = started (assuming no generators initially operational)
            model.addConstr(
                operational_type0[t] <= startup_type0[t],
                name=f"operational_startup_type0_{t}"
            )
            model.addConstr(
                operational_type1[t] <= startup_type1[t],
                name=f"operational_startup_type1_{t}"
            )
        else:
            # For subsequent periods: startups <= operational in previous period
            model.addConstr(
                startup_type0[t] <= operational_type0[t-1],
                name=f"max_startup_type0_{t}"
            )
            model.addConstr(
                startup_type1[t] <= operational_type1[t-1],
                name=f"max_startup_type1_{t}"
            )
            # Operational in current period <= operational in previous + started in current
            model.addConstr(
                operational_type0[t] <= operational_type0[t-1] + startup_type0[t],
                name=f"operational_startup_type0_{t}"
            )
            model.addConstr(
                operational_type1[t] <= operational_type1[t-1] + startup_type1[t],
                name=f"operational_startup_type1_{t}"
            )
        
        # Non-negativity
        model.addConstr(operational_type0[t] >= 0, name=f"nonneg_op_type0_{t}")
        model.addConstr(operational_type1[t] >= 0, name=f"nonneg_op_type1_{t}")
        model.addConstr(power_type0[t] >= 0, name=f"nonneg_power_type0_{t}")
        model.addConstr(power_type1[t] >= 0, name=f"nonneg_power_type1_{t}")
        model.addConstr(startup_type0[t] >= 0, name=f"nonneg_startup_type0_{t}")
        model.addConstr(startup_type1[t] >= 0, name=f"nonneg_startup_type1_{t}")
    
    model.optimize()
    
    if model.status == GRB.OPTIMAL:
        return {
            "objective": model.objVal,
            "status": "optimal"
        }
    elif model.status == GRB.INFEASIBLE:
        return {
            "objective": None,
            "status": "infeasible"
        }
    elif model.status == GRB.UNBOUNDED:
        return {
            "objective": None,
            "status": "unbounded"
        }
    else:
        return {
            "objective": None,
            "status": "failed"
        }

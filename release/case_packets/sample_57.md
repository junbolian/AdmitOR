import pyomo.environ as pyo
from pyomo.contrib.appsi.solvers.highs import Highs

def solve(params):
    # Extract parameters
    profit = params["profit"]  # list of 44 floats
    resource_coeff = params["resource_coeff"]  # list of 44 floats
    resource_capacity = params["resource_capacity"]  # float
    max_production = params["max_production"]  # list of 44 floats
    
    n_items = 44  # structural constant
    
    # Create model
    model = pyo.ConcreteModel()
    
    # Index set
    model.I = pyo.RangeSet(0, n_items - 1)
    
    # Decision variables: quantity of each item to produce
    model.x = pyo.Var(model.I, within=pyo.NonNegativeReals)
    
    # Objective: maximize total profit
    model.obj = pyo.Objective(
        expr=sum(profit[i] * model.x[i] for i in model.I),
        sense=pyo.maximize
    )
    
    # Resource constraint: total resource consumption <= capacity
    def resource_rule(model):
        return sum(resource_coeff[i] * model.x[i] for i in model.I) <= resource_capacity
    model.resource_con = pyo.Constraint(rule=resource_rule)
    
    # Production limits: x[i] <= max_production[i]
    def limit_rule(model, i):
        return model.x[i] <= max_production[i]
    model.limit_con = pyo.Constraint(model.I, rule=limit_rule)
    
    # Solve
    solver = Highs()
    result = solver.solve(model)
    
    # Check if solution is available via pyo.value
    try:
        obj_val = float(pyo.value(model.obj))
        return {'objective': obj_val, 'status': 'optimal'}
    except (ValueError, TypeError):
        # Infeasible or other failure
        return {'objective': None, 'status': 'infeasible or solve failed'}

from pyomo.contrib.appsi.solvers.highs import Highs


def solve(params):
    model = pyo.ConcreteModel()

    model.ITEMS = pyo.RangeSet(0, 43)

    profit_data = params["profit"]
    resource_coeff_data = params["resource_coeff"]
    resource_capacity_data = params["resource_capacity"]
    max_production_data = params["max_production"]

    model.profit = pyo.Param(model.ITEMS, initialize={item: profit_data[item] for item in range(44)})
    model.resource_coeff = pyo.Param(
        model.ITEMS, initialize={item: resource_coeff_data[item] for item in range(44)}
    )
    model.resource_capacity = pyo.Param(initialize=resource_capacity_data)
    model.max_production = pyo.Param(
        model.ITEMS, initialize={item: max_production_data[item] for item in range(44)}
    )

    model.x = pyo.Var(model.ITEMS, domain=pyo.NonNegativeReals)

    def resource_limit_rule(model):
        return sum(model.resource_coeff[item] * model.x[item] for item in model.ITEMS) <= model.resource_capacity

    model.resource_limit = pyo.Constraint(rule=resource_limit_rule)

    def production_limit_rule(model, item):
        return model.x[item] <= model.max_production[item]

    model.production_limit = pyo.Constraint(model.ITEMS, rule=production_limit_rule)

import gurobipy as gp
from gurobipy import GRB


def solve(params):
    profit = params["profit"]
    resource_coeff = params["resource_coeff"]
    resource_capacity = params["resource_capacity"]
    max_production = params["max_production"]

    n_items = 44

    model = gp.Model("production_optimization")
    model.setParam("OutputFlag", 0)

    # Decision variables: quantity of each item to produce
    x = model.addVars(n_items, lb=0.0, ub=[max_production[i] for i in range(n_items)], name="x")

    # Objective: maximize total profit
    model.setObjective(
        gp.quicksum(profit[i] * x[i] for i in range(n_items)),
        GRB.MAXIMIZE
    )

    # Resource constraint
    model.addConstr(
        gp.quicksum(resource_coeff[i] * x[i] for i in range(n_items)) <= resource_capacity,
        name="resource"
    )

    model.optimize()

    status = model.Status
    if status == GRB.OPTIMAL:
        return {"objective": model.ObjVal, "status": "optimal"}
    elif status == GRB.INFEASIBLE:
        return {"objective": None, "status": "infeasible"}
    elif status == GRB.UNBOUNDED:
        return {"objective": None, "status": "unbounded"}
    else:
        return {"objective": None, "status": f"gurobi_status_{status}"}

The total cost consists of a base cost of 75 units for each Type 0 generator and 79 units for each Type 1 generator that is operational in a given period, a variable cost of 4 units per megawatt of power generated by Type 0 generators and 3 units per megawatt for Type 1 generators, and a startup cost of 381 units for each Type 0 generator and 360 units for each Type 1 generator that is activated in a given period. 

The number of Type 0 generators operating in any period cannot exceed 3, and the number of Type 1 generators operating in any period cannot exceed 4. The total power generated by all generators in each period must meet or exceed the demand for that period, which is 165 MW in Period 0, 181 MW in Period 1, 203 MW in Period 2, and 216 MW in Period 3.

Similarly, the number of Type 1 generators starting up in Period 0 cannot exceed 1, and for subsequent periods, the number of startups is limited by the number of generators that were operational in the previous period. 

The variables representing the number of generators and the number of startups are general integer variables. The solution must determine the optimal number of generators to operate, the power output for each generator type, and the number of startups required in each period to minimize the total operational cost while ensuring that all demand and operational constraints are satisfied.

                       name=f"min_output_type1_{t}")
        
        # Maximum output constraints (per operational generator)
        model.addConstr(power_type0[t] <= max_output_type0 * n_operational_type0[t], 
                       name=f"max_output_type0_{t}")
        model.addConstr(power_type1[t] <= max_output_type1 * n_operational_type1[t], 
                       name=f"max_output_type1_{t}")
        
        # Buffer capacity constraint
        model.addConstr(
            max_output_type0 * n_operational_type0[t] + max_output_type1 * n_operational_type1[t] 
            >= demand[t] * (1 + buffer_factor),
            name=f"buffer_{t}"
        )
        
        # Startup constraints
        if t == 0:
            # Period 0: startups limited by maximum allowed
            model.addConstr(startups_type0[t] <= max_startups_type0_period0, 
                           name=f"max_startups_type0_{t}")
            model.addConstr(startups_type1[t] <= max_startups_type1_period0, 
                           name=f"max_startups_type1_{t}")
            # Operational generators in period 0 equal startups
            model.addConstr(n_operational_type0[t] == startups_type0[t], 
                           name=f"initial_operational_type0_{t}")
            model.addConstr(n_operational_type1[t] == startups_type1[t], 
                           name=f"initial_operational_type1_{t}")
        else:
            # Subsequent periods: startups limited by previous operational count
            model.addConstr(startups_type0[t] <= n_operational_type0[t-1], 
                           name=f"max_startups_type0_{t}")
            model.addConstr(startups_type1[t] <= n_operational_type1[t-1], 
                           name=f"max_startups_type1_{t}")
            # Operational count change relationship
            model.addConstr(n_operational_type0[t] <= n_operational_type0[t-1] + startups_type0[t], 
                           name=f"operational_change_type0_{t}")
            model.addConstr(n_operational_type1[t] <= n_operational_type1[t-1] + startups_type1[t], 
                           name=f"operational_change_type1_{t}")
        
        # Non-negativity
        model.addConstr(n_operational_type0[t] >= 0, name=f"nonneg_n_type0_{t}")
        model.addConstr(n_operational_type1[t] >= 0, name=f"nonneg_n_type1_{t}")
        model.addConstr(power_type0[t] >= 0, name=f"nonneg_power_type0_{t}")
        model.addConstr(power_type1[t] >= 0, name=f"nonneg_power_type1_{t}")
        model.addConstr(startups_type0[t] >= 0, name=f"nonneg_startups_type0_{t}")
        model.addConstr(startups_type1[t] >= 0, name=f"nonneg_startups_type1_{t}")
    
    model.optimize()
    
    if model.status == GRB.OPTIMAL:
        return {
            "objective": model.objVal,
            "status": "optimal"
        }
    elif model.status == GRB.INFEASIBLE:
        return {
            "objective": None,
            "status": "infeasible"
        }
    elif model.status == GRB.UNBOUNDED:
        return {
            "objective": None,
            "status": "unbounded"
        }
    else:
        return {
            "objective": None,
            "status": "failed"
        }
